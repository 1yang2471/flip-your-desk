```rust
/// 把 0..total 切成 ≤parts 个连续区间（并行分片用）；total=0 返回空。
fn capture_chunk_ranges(total: usize, parts: usize) -> Vec<std::ops::Range<usize>> {
    if total == 0 {
        return Vec::new();
    }
    let parts = parts.max(1).min(total);
    let base = total / parts;
    let rem = total % parts;
    (0..parts)
        .scan(0usize, |start, i| {
            let len = base + usize::from(i < rem);
            let range = *start..(*start + len);
            *start += len;
            Some(range)
        })
        .collect()
}
```

```rust
    monitors: &[MonitorInfo],
```

把原 icons 循环（windows.rs:1102-1134）整体替换为两阶段：

```rust
    // 阶段一（串行）：label_rect 走跨进程 SendMessage，多线程并发发消息
    // 收益小风险高——在进入并行区之前一次性收齐。
    let label_rects: Vec<Option<RectI>> = (0..items.len())
        .map(|slot| desktop.item_rect(slot, LVIR_LABEL).ok().flatten())
        .collect();

    // 阶段二（并行）：extract_cell / encode_png_data_url / tone_for_cell
    // 都是纯像素工作，输入区间互不重叠，可安全并行。
    let threads = std::thread::available_parallelism()
        .map(|n| n.get())
        .unwrap_or(4)
        .max(1);
    let ranges = capture_chunk_ranges(items.len(), threads);
    let chunks: Vec<Vec<IconRecord>> = std::thread::scope(|scope| {
        let handles: Vec<_> = ranges
            .into_iter()
            .map(|range| {
                let progress = &progress;
                scope.spawn(move || {
                    let mut chunk = Vec::with_capacity(range.len());
                    for slot in range {
                        let item = &items[slot];
                        let cell = RectI {
                            x: item.x,
                            y: item.y,
                            w: spacing.0,
                            h: spacing.1,
                        };
                        let monitor = geometry::monitor_for_cell(cell, monitors);
                        let label_rect = label_rects[slot];
                        let label_width = label_rect
                            .map(|rect| rect.right - rect.left)
                            .unwrap_or(cell.w);
                        let extract = extract_cell(&with_icons, &clean, origin, cell, label_width);
                        chunk.push(IconRecord {
                            id: slot as u32,
                            name: item.name.clone(),
                            image: extract
                                .image
                                .as_ref()
                                .and_then(|image| encode_png_data_url(image).ok())
                                .unwrap_or_default(),
                            cell,
                            monitor,
                            tone: tone_for_cell(&clean, origin, cell),
                            label: extract.label,
                        });
                    }
                    (range.start, chunk)
                })
            })
            .collect();
        handles
            .into_iter()
            .enumerate()
            .map(|(index, handle)| {
                let (start, chunk) = handle.join().unwrap();
                progress(start, items.len());
                chunk
            })
            .collect::<Vec<_>>()
    });
    let icons: Vec<IconRecord> = chunks.into_iter().flatten().collect();
```

注意：`chunks` 的收集顺序必须保持槽位顺序——ranges 按 slot 升序生成、join 按提交顺序收
集即可保证；末尾 flatten 得到 id 连续的 icons。
